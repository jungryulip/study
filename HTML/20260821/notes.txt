	<title>Array 객체의 메서드</title>
	<style>
    body {
      display: flex;
      flex-direction: column;
      justify-content: center;
      align-items: center;
      height: 100vh;
			font-size: 1.2em;
    }
		hr {
			width: 100%;			
		}
	</style>
</head>
<body>
	<div class="container">
		<script>
			let nums = [1, 2, 3];
			let chars = ["a", "b", "c", "d"];
	
			// 요소 추가 - 새로운 length 값 반환
			let ret1 = nums.push(4,5); // 배열 끝에 추가
			let ret2 = chars.unshift(0); // 배열 앞에 추가
			console.log(`새 배열 : ${nums}, 반환값 : ${ret1}`)
			console.log(`새 배열 : ${chars}, 반환값 : ${ret2}`)
	
			// 요소 추출 - 꺼낸 요소 반환
			let pop1 = chars.pop(); //마지막 요소 꺼냄
			console.log(`꺼낸 요소 : ${pop1}, chars : ${chars}`);
			let pop2 = chars.shift(); // 첫 번째 요소 꺼냄
			console.log(`꺼낸 요소 : ${pop2}, chars : ${chars}`);
	
		</script>
	</div>
</body>
</html>

      // Do it! 인수 개수에 따라 splice 메서드 사용하기
      let study = ['html', 'css', 'web', 'jquery', 'react', 'nodejs'];
      document.write(` 처음 상태 : ${study}<br>`);
      document.write(`<hr>`)
      // 인수가 1개일 경우
      let js = study.splice(2); //2부터 끝까지 삭제
      document.write(`반환된 배열 : ${js}<br>`);
      document.write(`변경된 배열 : ${study}<br>`);
      document.write(`<hr>`)

      // 인수가 2개일 경우- 시작위치, 삭제할 갯수
      let jquery = js.splice(1,1);
      document.write(`반환된 배열 : ${jquery}<br>`);
      document.write(`변경된 배열 : ${js}<br>`);
      document.write(`<hr>`)

      // 인수가 3개 이상일 경우-시작위치, 삭제할 갯수, 삭제할 위치에 들어갈 값
      let mjquery = js.splice(1,0,'typescript');
      document.write(`반환된 배열 : ${mjquery}<br>`);
      document.write(`변경된 배열 : ${js}<br>`);
      document.write(`<hr>`)
      
    </script>
  </div>
</body>
</html>
