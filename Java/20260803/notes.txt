package chap04;

public class _01_IfExample1 {
    public static void main(String[] args) {
        // 조건문 If
        int age = 10;

        // if문 내에서 하나의 문장을 실험할때는 {}를 생략 가능
        // 가독성을 위해서 if for 반복문에서도 습관적으로 중괄호를 쓰는 습관을 들이세요.
        if (age>=8){
            System.out.println("학교에 다닙니다.");
        }
        System.out.println("-----");
        if (age>=8)
            System.out.println("학교에 다닙니다.");
        System.out.println("즐거운 학교 생활 하세요!");//항살 실행됨
        // if문에서 수행할 문장이 한개라면 {}생략이 가능하지만, 한개 이상의 문장을 실행할때는 {}로가 필요하다.
        // 항상 실행되는 문장을 출력하고 싶다면 if 조건문의 {}밖에서 적으면 된다.
        System.out.println("-----");
        if (age>=8){
            System.out.println("학교에 다닙니다.");
            System.out.println("즐거운 학교 생활 하세요!");
        }
        System.out.println("그럼 안녕!");

    }
}

package chap04;

public class _04_IfExample4 {

    public static void main(String[] args) {
        // 나이에 따른 입장료 계산
        // if-else문은 조건을 순서대로 검사하며, 처음으로 참true인 조건의 코드만 실행한 후 if-else문을 종료
        // 각 if문은 서로 연결되어 있지 않으므로 순서대로 검사한다. 
        int age = 9; // 나이

        if (age < 14) { //true
            charge = 2000;
            System.out.println("초등학생입니다.");
        }
        if (age < 20) {
            charge = 2500;
            System.out.println("중, 고등학생입니다.");
        }
        else {
            charge = 3000;
            System.out.println("일반인입니다.");
        }
        System.out.println("입장료는 " + charge + "원 입니다.");// 최종 계산된 입장료
    }
}

package chap04;

public class _05_SwitchCase {
    public static void main(String[] args) {
        // switch 문은 하나의 값에 여러 경우 case 둥 하나를 선택하여 실행
        // 값이 간단한 점수, 문자, 문자열 등을 비교할때 if-else문보다 코드가 간결하게 잣겅할 수 있다.
        // break는 현재case의 실행을 종료하고 switch문을 빠져나간다.
        // switch문 전체에서 중괄호{}는 한번만 사용한다.
        int ranking =1;
        char medalColor;

        switch (ranking) {
            //ranking =1 금메달
            case 1: medalColor = 'G';
            break; // break를 사용하지 않으면 멈추지 않고 다음으로 넘어간다.
            case 2: medalColor = 'S';
            break;
            case 3: medalColor = 'B';
            break;
            default:
                medalColor = 'A';
        }
        System.out.println(ranking + "등 메달의 색깔은 " + medalColor + "입니다.");
    }
}

package chap04;

public class _08_SwitchCase5 {
    public static void main(String[] args) {
        int month = 13;
        int day = switch (month) {
            //31일
            case 1, 3, 5, 7, 8, 10, 12 -> 31;
            case 4,6,9,11-> 30;
            case 2-> 28;
            default -> {
                if (month < 1 || month > 12) {
                    System.out.println("없는 달입니다.");
                } else { //예외적인 상황
                    System.out.println("알 수 없는 오류입니다.");
                }
                yield 0;
            }
        };
        System.out.println(month + "월의 날짜는 " + day + "일까지 있습니다.");
    }
}

package chap04.loopeexample;

public class _03_DoWhileExample {
    public static void main(String[] args) {
        // do while문은 수행문을 실행하고 나중에 조건은 검사
        // 조건이 처음부처 거짓이라고 하더라도 최소 1번은 수행됨
        // for, while문이 많이 사용되고, do while은 상대적으로 사용빈도가 낮습니다.
        // while = 조건확인 -> 실행
        int num = 20;
        int sum = 0;

        do {
            sum += num;
            num++;
        } while (num <= 10);
        System.out.println("1부터 10까지의 합은 " + sum + "입니다.");
    }
}
