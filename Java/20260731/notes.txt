package chap02;

public class _07_TypeCasting {
    public static void main(String[] args) {

        // 실수형에서 정수형으로

package chap03;

public class _01_OperationEx {
    public static void main(String[] args) {
        //대입연산자

        int matScore = 90;
        int engScore = 70;

        int totalScore = matScore + engScore;
        System.out.println(totalScore);

        double avgScore = totalScore / 2.0;
        System.out.println(avgScore);

        System.out.println("----");

        // % 나머지 연산자
        int number = 10;
        int quotient = number / 3;
        int remainder = number % 3;

        System.out.println("몫 = " + quotient);
        System.out.println("나머지 = " + remainder);

        // 부호 연산자
        int num = 10;
        int num2 = -num;
        System.out.println(num);
        System.out.println(num2);

        num = -num;
        System.out.println(num);

        System.out.println("----");

        // 복합 대입연산자
        num = 10;
//        num = num + 2; // num += 2와 결과값이 같다.
        num += 2;
        System.out.println(num);//12

        // num num-2;
        num -= 2;
        System.out.println(num);//10

        // num * 2 = num
        num *= 2;
        System.out.println(num);//20

        // num /2 = num
        num /= 2;
        System.out.println(num);//10

        // num = num % 2
        num %= 2;
        System.out.println(num);//0
    }

증가감소 연산자[전위연산자 후위연산자]
값에++나 --의 위치에 따라 값이 달라진다. 
package chap03;

public class _02_OperationEx2 {
    public static void main(String[] args) {

        // 증감 연산자 ++,--
        int gameScore = 150;
        System.out.println(gameScore); //150
        System.out.println(++gameScore); //151 전위연산자
        System.out.println(gameScore); //151

        gameScore = 150;
        System.out.println(gameScore); //150
        System.out.println(gameScore++); //150 후위연산자
        System.out.println(gameScore); //151

        System.out.println("----");

        gameScore = 150;
        System.out.println(gameScore); //150
        System.out.println(--gameScore); //149
        System.out.println(gameScore); //149

        gameScore = 150;
        System.out.println(gameScore); //150
        System.out.println(gameScore--); //150
        System.out.println(gameScore); //149

        System.out.println("----");

        // 식당 대기번호 표
        int waiting = 0;
        System.out.println("대기 인원 : "+waiting++); //0
        System.out.println("대기 인원 : "+waiting++); //1
        System.out.println("대기 인원 : "+waiting++); //2
        System.out.println("대기 인원 : "+waiting); //3


    }
}

package chap03;

public class _03_OperationEx3 {
    public static void main(String[] args) {
        // 논리연산자
        int num1 = 100;
        int i = 2;

        // and 연산자
        boolean value = ((num1 = num1 + 10) < 10) && ((i = i + 2) < 10);
        System.out.println(value); // false 110<10 && 4<10
        System.out.println(num1); // 110
        System.out.println(i); // 2

        // or 연산자
        value = ((num1 = num1 + 10) < 10) || ((i = i + 2) < 10);
        System.out.println(value); // true 110<10 && 4<10
        System.out.println(num1); // 120
        System.out.println(i); // 4

        // 논리 부정 연산자
        System.out.println(!true);
        System.out.println(!false);
        System.out.println(!(5 == 5));
        System.out.println(!(5 == 3));

    }
}
