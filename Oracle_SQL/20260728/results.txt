/*문제1[시퀀스를 먼저 생성하고 만들면 employee_id를 1씩 증가하게 만들 수 있다. ]
아래와 같이 EMPLOYEE테이블을 생성하세요.
EMPLOYE_ID는 PRIMARY KEY, NAME은 NOT NULL
Name        Null        Type
-------- ------------- ------------
EMPLOYEE_ID             NUMBER(7)
NAME                    VARCHAR2(25)
ADDRESS                 VARCHAR2(25)
DEPTNO                  NUMBER(2)
PHONE_NUMBER            VARCHAR2(20)
*/
create sequence emp_seq
increment by 1
start with 1
maxvalue 9999999;

--Table EMPLOYEE이(가) 생성되었습니다.
create table employee (
    employee_id  number(7) default emp_seq.nextval,
    name         varchar2(25) not null,
    address      varchar2(25),
    deptno       number(2),
    phone_number varchar2(20),
    constraint employee_pk primary key(employee_id)
);
--Table EMPLOYEE이(가) 생성되었습니다.
